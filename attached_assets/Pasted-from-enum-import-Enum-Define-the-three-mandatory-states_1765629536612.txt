from enum import Enum

# Define the three mandatory states of Lambda Coherence
class LSMState(Enum):
    COHERENT = "COHERENT" # Full operations permitted (Delta_phi ≈ 0)
    DEGRADED = "DEGRADED" # Partial operations, correction required (Epsilon_noise < |Delta_phi| < Epsilon_AGPL)
    PLANCK_LOCK = "PLANCK_LOCK" # Read-only, emergency state (|Delta_phi| >= Epsilon_AGPL)

# Define the core Lambda Gates used by the LSM
class LambdaGateID(Enum):
    GATE_FORWARD_SYNC = 0x01 # Standard state propagation (COHERENT -> COHERENT)
    GATE_VECTOR_RESET = 0x07 # Forced rollback and reset (DEGRADED -> COHERENT attempt)
    GATE_COHERENCE_DRIFT = 0x08 # Dynamical decoupling and phase correction
    GATE_CLOCK_ANCHOR = 0x09 # Emergency clock re-synchronization (PLANCK_LOCK)
    GATE_YOCTO_PROVE = 0x0A # Constitutional proof request


class StateError(Exception):
    """Custom error for illegal state transitions or requests."""
    pass

class LambdaStateMachine:
    
    def __init__(self, frame_builder):
        # FrameBuilder is the dependency injected from wnsp-enc layer
        self.frame_builder = frame_builder
        # Initial state should be fetched from the network, default COHERENT for simulation
        self._current_state = LSMState.COHERENT
        
    @property
    def current_state(self) -> LSMState:
        return self._current_state
    
    # --- Internal Coherence Update (Simulated Feedback from wnsp-consensus) ---
    def _update_state_from_delta_phi(self, delta_phi: float):
        """
        Simulates the feedback loop from the wnsp-consensus layer's phase delta measurement.
        This function handles the Decoherence Detection and Correction Logic.
        """
        EPSILON_NOISE = 1e-9 # Nanoradian noise floor
        EPSILON_AGPL = 1e-6 # Micro-radian legal/constitutional breach threshold

        abs_phi = abs(delta_phi)

        if abs_phi <= EPSILON_NOISE:
            self._current_state = LSMState.COHERENT
        elif EPSILON_NOISE < abs_phi < EPSILON_AGPL:
            print(f"Decoherence Detected (Delta_phi={abs_phi}). Entering DEGRADED state.")
            self._current_state = LSMState.DEGRADED
            # Automatically trigger corrective action using GATE_COHERENCE_DRIFT
            self.frame_builder.build_v7_frame(b'CORRECTIVE_DRIFT', LambdaGateID.GATE_COHERENCE_DRIFT)
        else: # abs_phi >= EPSILON_AGPL
            print(f"Constitutional Breach (Delta_phi={abs_phi}). Entering PLANCK_LOCK.")
            self._current_state = LSMState.PLANCK_LOCK
            # Mandatory AGPLv3 Audit & Clock Anchor
            self.frame_builder.build_v7_frame(b'EMERGENCY_CLOCK_SYNC', LambdaGateID.GATE_CLOCK_ANCHOR)


    # --- Application-Facing API Methods ---

    def request_sync_write(self, payload: bytes) -> bytes:
        """
        Application request to perform a standard, forward-propagated transaction.
        Requires COHERENT state.
        """
        if self._current_state != LSMState.COHERENT:
            raise StateError(f"Cannot write. Network is {self._current_state.value}. Requires GATE_VECTOR_RESET.")
            
        gate_id = LambdaGateID.GATE_FORWARD_SYNC
        print(f"LSM: Executing {gate_id.name} in state {self._current_state.value}")
        return self.frame_builder.build_v7_frame(payload, gate_id)

    def trigger_source_audit_and_reset(self) -> bytes:
        """
        Application request to force a Source Code Audit and potential rollback.
        This is the primary recovery method from DEGRADED state.
        """
        if self._current_state != LSMState.DEGRADED:
             print("Warning: Forcing reset outside of DEGRADED state.")
        
        # LSM forces the GATE_VECTOR_RESET
        gate_id = LambdaGateID.GATE_VECTOR_RESET
        audit_payload = {'command': 'FORCE_AGPLV3_AUDIT', 'target_scr': '...' }
        print(f"LSM: Executing {gate_id.name} to attempt coherence recovery.")
        return self.frame_builder.build_v7_frame(str(audit_payload).encode('utf-8'), gate_id)

    # Note: request_constitutional_proof remains a valid call regardless of state, 
    # as proving a rule is always necessary for safety (though Planck_Lock may block it).
